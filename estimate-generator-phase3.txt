                  keyboardType="decimal-pad"

                <Text style={{ color: Colors.sub, fontSize: 12, marginTop: 4 }}>
                  Current markup: {money(calc.profit)} ({bid.markupPct || 0}%)
                </Text>
              </View>
            </View>
          </ScrollView>
        );
      }
      
      case 7: {
        return (
          <ProjectAnalysis
            bid={bid}
            calc={calc}
            materialsCart={materialsCart}
            laborLineItems={bid.laborLineItems || []}
          />
        );
      }
      
      case 8: {
        const scheduleType = bid.paymentSchedule || 'milestone-based';
        const milestones = bid.paymentMilestones || [];
        const weeklyPayments = bid.weeklyPayments || [];
        const grandTotal = calc?.grandTotal || calc?.total || 0;
        
        // Calculate totals
        const milestoneTotal = milestones.reduce((sum, m) => sum + (m.paymentAmount || 0), 0);
        const weeklyTotal = weeklyPayments.reduce((sum, w) => sum + (w.amount || 0), 0);
        const milestoneTotalPct = milestones.reduce((sum, m) => sum + (m.percentage || 0), 0);
        
        return (
          <ScrollView style={{ flex: 1 }} showsVerticalScrollIndicator={false}>
            <View style={{ padding: 16 }}>
              <View style={s.inputGroup}>
                <Text style={s.label}>Payment Schedule Type</Text>
                <View style={{ flexDirection: 'row', marginBottom: 16 }}>
                  <TouchableOpacity
                    style={[s.chip, scheduleType === 'milestone-based' && s.chipActive]}
                    onPress={() => {
                      updateBid('paymentSchedule', 'milestone-based');
                      Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light);
                    }}
                  >
                    <Text style={[s.chipText, scheduleType === 'milestone-based' && { color: '#fff' }]}>
                      Milestone-Based
                    </Text>
                  </TouchableOpacity>
                  <TouchableOpacity
                    style={[s.chip, scheduleType === 'weekly' && s.chipActive]}
                    onPress={() => {
                      updateBid('paymentSchedule', 'weekly');
                      Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light);
                    }}
                  >
                    <Text style={[s.chipText, scheduleType === 'weekly' && { color: '#fff' }]}>
                      Weekly
                    </Text>
                  </TouchableOpacity>
                </View>
              </View>
              
              {scheduleType === 'milestone-based' ? (
                <View>
                  <View style={{ flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center', marginBottom: 12 }}>
                    <Text style={[s.label, { marginBottom: 0 }]}>Payment Milestones</Text>
                    <TouchableOpacity
                      style={{
                        flexDirection: 'row',
                        alignItems: 'center',
                        backgroundColor: Colors.primary,
                        paddingHorizontal: 12,
                        paddingVertical: 6,
                        borderRadius: 8,
                      }}
                      onPress={handleAddMilestone}
                    >
                      <Ionicons name="add" size={16} color="#fff" />
                      <Text style={{ color: '#fff', fontSize: 12, fontWeight: '600', marginLeft: 4 }}>
                        Add Milestone
                      </Text>
                    </TouchableOpacity>
                  </View>
                  
                  {milestones.length === 0 ? (
                    <View style={[s.stepCard, { padding: 24, alignItems: 'center' }]}>
                      <Ionicons name="calendar-outline" size={48} color={Colors.sub} />
                      <Text style={{ color: Colors.sub, fontSize: 14, marginTop: 12, textAlign: 'center' }}>
                        No payment milestones added yet.{'\n'}Add milestones to track payment schedule.
                      </Text>
                    </View>
                  ) : (
                    milestones.map((milestone, index) => {
                      // Calculate percentage if not set but amount is
                      let displayPercentage = milestone.percentage || 0;
                      if (!displayPercentage && milestone.paymentAmount && grandTotal > 0) {
                        displayPercentage = Math.round((milestone.paymentAmount / grandTotal) * 100);
                      }
                      
                      return (
                        <View key={milestone.id || index} style={[s.stepCard, { marginBottom: 12, position: 'relative' }]}>
                          <TouchableOpacity
                            style={{
                              position: 'absolute',
                              top: 12,
                              right: 12,
                              flexDirection: 'row',
                              gap: 8,
                            }}
                            onPress={() => handleEditMilestone(milestone)}
                          >
                            <TouchableOpacity
                              onPress={() => handleEditMilestone(milestone)}
                              style={{ padding: 4 }}
                            >
                              <Ionicons name="create-outline" size={18} color={Colors.primary} />
                            </TouchableOpacity>
                            <TouchableOpacity
                              onPress={() => handleDeleteMilestone(milestone.id)}
                              style={{ padding: 4 }}
                            >
                              <Ionicons name="trash-outline" size={18} color={Colors.red} />
                            </TouchableOpacity>
                          </TouchableOpacity>
                          
                          <Text style={{ color: Colors.text, fontSize: 16, fontWeight: '600', marginBottom: 8, paddingRight: 60 }}>
                        {milestone.name || `Milestone ${index + 1}`}
                      </Text>
                          {milestone.description && (
                            <Text style={{ color: Colors.sub, fontSize: 12, marginBottom: 8 }}>
                              {milestone.description}
                            </Text>
                          )}
                      <View style={{ flexDirection: 'row', justifyContent: 'space-between', marginBottom: 8 }}>
                            <View>
                              <Text style={{ color: Colors.sub, fontSize: 11, marginBottom: 2 }}>Amount</Text>
                              <Text style={{ color: Colors.text, fontSize: 16, fontWeight: '600' }}>
                                {money(milestone.paymentAmount || 0)}
                              </Text>
                            </View>
                            <View style={{ alignItems: 'flex-end' }}>
                              <Text style={{ color: Colors.sub, fontSize: 11, marginBottom: 2 }}>Percentage</Text>
                              <Text style={{ color: Colors.text, fontSize: 16, fontWeight: '600' }}>
                                {displayPercentage}%
                              </Text>
                            </View>
                      </View>
                      {milestone.scheduledDate && (
                            <View style={{ flexDirection: 'row', alignItems: 'center', marginTop: 8 }}>
                              <Ionicons name="calendar-outline" size={14} color={Colors.sub} />
                              <Text style={{ color: Colors.sub, fontSize: 12, marginLeft: 6 }}>
                                Due: {milestone.scheduledDate}
                        </Text>
                            </View>
                      )}
                    </View>
                      );
                    })
                  )}
                  
                  {milestones.length > 0 && (
                    <View style={[s.stepCard, { backgroundColor: 'rgba(34, 197, 94, 0.1)', borderColor: 'rgba(34, 197, 94, 0.3)' }]}>
                      <View style={{ flexDirection: 'row', justifyContent: 'space-between', marginBottom: 4 }}>
                        <Text style={{ color: Colors.sub, fontSize: 12 }}>Total Scheduled</Text>
                        <Text style={{ color: Colors.text, fontSize: 14, fontWeight: '600' }}>
                          {money(milestoneTotal)}
                        </Text>
                      </View>
                      <View style={{ flexDirection: 'row', justifyContent: 'space-between' }}>
                        <Text style={{ color: Colors.sub, fontSize: 12 }}>Total Percentage</Text>
                        <Text style={{ color: Colors.text, fontSize: 14, fontWeight: '600' }}>
                          {milestoneTotalPct}%
                        </Text>
                      </View>
                      {grandTotal > 0 && Math.abs(milestoneTotal - grandTotal) > 1 && (
                        <Text style={{ color: Colors.orange, fontSize: 11, marginTop: 8 }}>
                          ⚠️ Total doesn't match bid amount ({money(grandTotal)})
                        </Text>
                      )}
                    </View>
                  )}
                </View>
              ) : (
                <View>
                  <View style={{ flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center', marginBottom: 12 }}>
                    <Text style={[s.label, { marginBottom: 0 }]}>Weekly Payments</Text>
                    <TouchableOpacity
                      style={{
                        flexDirection: 'row',
                        alignItems: 'center',
                        backgroundColor: Colors.primary,
                        paddingHorizontal: 12,
                        paddingVertical: 6,
                        borderRadius: 8,
                      }}
                      onPress={handleAddWeeklyPayment}
                    >
                      <Ionicons name="add" size={16} color="#fff" />
                      <Text style={{ color: '#fff', fontSize: 12, fontWeight: '600', marginLeft: 4 }}>
                        Add Payment
                      </Text>
                    </TouchableOpacity>
                  </View>
                  
                  {weeklyPayments.length === 0 ? (
                    <View style={[s.stepCard, { padding: 24, alignItems: 'center' }]}>
                      <Ionicons name="calendar-outline" size={48} color={Colors.sub} />
                      <Text style={{ color: Colors.sub, fontSize: 14, marginTop: 12, textAlign: 'center' }}>
                        No weekly payments added yet.{'\n'}Add weekly payments to track progress payments.
                      </Text>
                    </View>
                  ) : (
                    weeklyPayments.map((payment, index) => {
                      // Calculate percentage if not set but amount is
                      let displayPercentage = payment.percentage || 0;
                      if (!displayPercentage && payment.amount && grandTotal > 0) {
                        displayPercentage = Math.round((payment.amount / grandTotal) * 100);
                      }
                      
                      return (
                        <View key={payment.id || index} style={[s.stepCard, { marginBottom: 12, position: 'relative' }]}>
                          <TouchableOpacity
                            style={{
                              position: 'absolute',
                              top: 12,
                              right: 12,
                              flexDirection: 'row',
                              gap: 8,
                            }}
                          >
                            <TouchableOpacity
                              onPress={() => handleEditWeeklyPayment(payment)}
                              style={{ padding: 4 }}
                            >
                              <Ionicons name="create-outline" size={18} color={Colors.primary} />
                            </TouchableOpacity>
                            <TouchableOpacity
                              onPress={() => handleDeleteWeeklyPayment(payment.id)}
                              style={{ padding: 4 }}
                            >
                              <Ionicons name="trash-outline" size={18} color={Colors.red} />
                            </TouchableOpacity>
                          </TouchableOpacity>
                          
                          <Text style={{ color: Colors.text, fontSize: 16, fontWeight: '600', marginBottom: 8, paddingRight: 60 }}>
                            {payment.description || `Week ${payment.weekNumber || index + 1} Payment`}
                      </Text>
                      <View style={{ flexDirection: 'row', justifyContent: 'space-between', marginBottom: 8 }}>
                            <View>
                              <Text style={{ color: Colors.sub, fontSize: 11, marginBottom: 2 }}>Amount</Text>
                              <Text style={{ color: Colors.text, fontSize: 16, fontWeight: '600' }}>
                                {money(payment.amount || 0)}
                              </Text>
                            </View>
                            <View style={{ alignItems: 'flex-end' }}>
                              <Text style={{ color: Colors.sub, fontSize: 11, marginBottom: 2 }}>Percentage</Text>
                              <Text style={{ color: Colors.text, fontSize: 16, fontWeight: '600' }}>
                                {displayPercentage}%
                              </Text>
                            </View>
                      </View>
                      {payment.scheduledDate && (
                            <View style={{ flexDirection: 'row', alignItems: 'center', marginTop: 8 }}>
                              <Ionicons name="calendar-outline" size={14} color={Colors.sub} />
                              <Text style={{ color: Colors.sub, fontSize: 12, marginLeft: 6 }}>
                                Due: {payment.scheduledDate}
                        </Text>
                            </View>
                      )}
                    </View>
                      );
                    })
                  )}
                  
                  {weeklyPayments.length > 0 && (
                    <View style={[s.stepCard, { backgroundColor: 'rgba(34, 197, 94, 0.1)', borderColor: 'rgba(34, 197, 94, 0.3)' }]}>
                      <View style={{ flexDirection: 'row', justifyContent: 'space-between' }}>
                        <Text style={{ color: Colors.sub, fontSize: 12 }}>Total Scheduled</Text>
                        <Text style={{ color: Colors.text, fontSize: 14, fontWeight: '600' }}>
                          {money(weeklyTotal)}
                        </Text>
                      </View>
                      {grandTotal > 0 && Math.abs(weeklyTotal - grandTotal) > 1 && (
                        <Text style={{ color: Colors.orange, fontSize: 11, marginTop: 8 }}>
                          ⚠️ Total doesn't match bid amount ({money(grandTotal)})
                        </Text>
                      )}
                    </View>
                  )}
                </View>
              )}
            </View>
          </ScrollView>
        );
      }
      
      case 9: {
        return (
          <ScrollView style={{ flex: 1 }} showsVerticalScrollIndicator={false}>
            <View style={{ padding: 16 }}>
              <View style={s.inputGroup}>
                <View style={{ flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center', marginBottom: 12 }}>
                  <Text style={s.label}>License Required</Text>
                  <Switch
                    value={bid.license || false}
                    onValueChange={(value) => updateBid('license', value)}
                    trackColor={{ false: Colors.line, true: Colors.primary }}
                    thumbColor={bid.license ? '#fff' : Colors.sub}
                  />
                </View>
              </View>
              
              <View style={s.inputGroup}>
                <View style={{ flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center', marginBottom: 12 }}>
                  <Text style={s.label}>Insurance Required</Text>
                  <Switch
                    value={bid.insurance || false}
                    onValueChange={(value) => updateBid('insurance', value)}
                    trackColor={{ false: Colors.line, true: Colors.primary }}
                    thumbColor={bid.insurance ? '#fff' : Colors.sub}
                  />
                </View>
              </View>
              
              <View style={s.inputGroup}>
                <View style={{ flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center', marginBottom: 12 }}>
                  <Text style={s.label}>Bond Required</Text>
                  <Switch
                    value={bid.bond || false}
                    onValueChange={(value) => updateBid('bond', value)}
                    trackColor={{ false: Colors.line, true: Colors.primary }}
                    thumbColor={bid.bond ? '#fff' : Colors.sub}
                  />
                </View>
              </View>
              
              <View style={s.inputGroup}>
                <View style={{ flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center', marginBottom: 12 }}>
                  <Text style={s.label}>OSHA Compliance</Text>
                  <Switch
                    value={bid.osha || false}
                    onValueChange={(value) => updateBid('osha', value)}
                    trackColor={{ false: Colors.line, true: Colors.primary }}
                    thumbColor={bid.osha ? '#fff' : Colors.sub}
                  />
                </View>
              </View>
            </View>
          </ScrollView>
        );
      }
      
      case 10: {
        const aiLevel = healthScore >= 75 ? 'good' : healthScore >= 55 ? 'warn' : 'risk';
        
        return (
          <ScrollView style={{ flex: 1 }} showsVerticalScrollIndicator={false}>
            <View style={{ padding: 16 }}>
              <LinearGradient
                colors={["rgba(34, 197, 94, 0.15)", "rgba(34, 211, 238, 0.08)", "rgba(34, 197, 94, 0)"]}
                start={{ x: 0, y: 0 }}
                end={{ x: 1, y: 1 }}
                style={{
                  borderRadius: 20,
                  padding: 20,
                  marginBottom: 16,
                  borderWidth: 1,
                  borderColor: 'rgba(34, 197, 94, 0.3)',
                  alignItems: 'center',
                }}
              >
                <Text style={{ color: Colors.sub, fontSize: 12, marginBottom: 8, fontWeight: '600', letterSpacing: 1 }}>
                  HEALTH SCORE
                </Text>
                <Text style={{ color: healthColor, fontSize: 48, fontWeight: '700', marginBottom: 8 }}>
                  {healthScore}
                </Text>
                <Text style={{ color: Colors.sub, fontSize: 14 }}>
                  {aiLevel === 'good' ? 'Ready to send' : aiLevel === 'warn' ? 'Review recommended' : 'Needs attention'}
                </Text>
              </LinearGradient>
              
              <TouchableOpacity
                style={{
                  backgroundColor: Colors.primary,
                  borderRadius: 12,
                  padding: 16,
                  alignItems: 'center',
                  marginBottom: 16,
                }}
                onPress={generateContract}
              >
                <Text style={{ color: '#fff', fontSize: 16, fontWeight: '700' }}>
                  Generate & Export Contract
                </Text>
              </TouchableOpacity>
              
              <TouchableOpacity
                style={s.outlineBtn}
                onPress={handleSaveEstimate}
              >
                <Text style={s.outlineBtnText}>Save Bid</Text>
              </TouchableOpacity>
            </View>
          </ScrollView>
        );
      }
      
      default:
        return null;
    }
  };

  return (
    <SafeAreaView style={s.container} edges={['top']}>
      <StatusBar barStyle="light-content" />
      
      {/* Header */}
      <View style={{ paddingHorizontal: 16, paddingTop: 8, paddingBottom: 12 }}>
        <View style={{ flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center', marginBottom: 16 }}>
          <Text style={{ color: Colors.text, fontSize: 28, fontWeight: '800' }}>Bid Builder</Text>
          <TouchableOpacity
            style={{
              backgroundColor: Colors.primary,
              paddingHorizontal: 16,
              paddingVertical: 8,
              borderRadius: 20,
            }}
            onPress={() => {
              setBid(blankState());
              setMaterialsCart([]);
              setStep(1);
            }}
          >
            <Text style={{ color: '#000', fontSize: 14, fontWeight: '700' }}>+ New</Text>
          </TouchableOpacity>
        </View>
        
        {/* Navigation Pill - bright green border */}
        <View style={{ 
          borderRadius: 30, 
          borderWidth: 1, 
          borderColor: '#19E180',
          marginBottom: 16,
        }}>
          <View style={{ flexDirection: 'row', backgroundColor: '#000000', borderRadius: 28, padding: 4 }}>
            <TouchableOpacity
              onPress={() => {
                if (step > 1) {
                  setStep(step - 1);
                } else {
                  router.back();
                }
              }}
              style={{ paddingHorizontal: 16, paddingVertical: 10, borderRadius: 24, backgroundColor: 'rgba(255,255,255,0.05)' }}
            >
              <Text style={{ color: Colors.text, fontSize: 14, fontWeight: '600' }}>← Back</Text>
            </TouchableOpacity>
            <TouchableOpacity
              onPress={() => setShowCostBreakdownModal(true)}
              style={{ flex: 1, paddingHorizontal: 16, paddingVertical: 10, borderRadius: 24, alignItems: 'center' }}
            >
              <Text style={{ color: Colors.text, fontSize: 14, fontWeight: '600' }}>View Summary</Text>
            </TouchableOpacity>
            <LinearGradient
              colors={['#2DFFC4', '#00A6FF']}
              start={{ x: 0, y: 0.5 }}
              end={{ x: 1, y: 0.5 }}
              style={{ borderRadius: 24, paddingHorizontal: 16, paddingVertical: 10 }}
            >
              <TouchableOpacity
                onPress={() => {
                  if (step < 10) {
                    setStep(step + 1);
                  }
                }}
              >
                <Text style={{ color: '#000', fontSize: 14, fontWeight: '700' }}>Next →</Text>
              </TouchableOpacity>
            </LinearGradient>
          </View>
        </View>
        
        {/* Step Section Card with Icons - bright green border */}
        <View style={{ 
          borderRadius: 20, 
          borderWidth: 1, 
          borderColor: '#19E180',
        }}>
          <View style={{ backgroundColor: '#000000', borderRadius: 18, paddingVertical: 12, paddingHorizontal: 8 }}>
            {/* Current Step Info */}
            <View style={{ flexDirection: 'row', alignItems: 'center', justifyContent: 'center', marginBottom: 12 }}>
              <View style={{ width: 36, height: 36, borderRadius: 18, backgroundColor: 'rgba(25, 225, 128, 0.2)', justifyContent: 'center', alignItems: 'center', marginRight: 10 }}>
                <MaterialIcons name={getStepIcon(step)} size={20} color="#19E180" />
              </View>
              <View>
                <Text style={{ color: Colors.text, fontSize: 16, fontWeight: '700' }}>{STEPS[step - 1]?.title}</Text>
                <Text style={{ color: Colors.sub, fontSize: 12 }}>{STEPS[step - 1]?.subtitle}</Text>
              </View>
            </View>
            
            {/* Step Icons Row */}
            <ScrollView horizontal showsHorizontalScrollIndicator={false} contentContainerStyle={{ paddingHorizontal: 4 }}>
              {STEPS.map((stepItem) => (
                <TouchableOpacity
                  key={stepItem.id}
                  onPress={() => setStep(stepItem.id)}
                  style={{
                    alignItems: 'center',
                    marginHorizontal: 6,
                    opacity: step === stepItem.id ? 1 : 0.5,
                  }}
                >
                  <View style={{
                    width: 40,
                    height: 40,
                    borderRadius: 20,
                    backgroundColor: step === stepItem.id ? 'rgba(25, 225, 128, 0.25)' : 'rgba(255, 255, 255, 0.05)',
                    borderWidth: step === stepItem.id ? 2 : 1,
                    borderColor: step === stepItem.id ? '#19E180' : 'rgba(255, 255, 255, 0.1)',
                    justifyContent: 'center',
                    alignItems: 'center',
                    marginBottom: 4,
                  }}>
                    <MaterialIcons
                      name={getStepIcon(stepItem.id)}
                      size={18}
                      color={step === stepItem.id ? '#19E180' : Colors.sub}
                    />
                  </View>
                  <Text style={{
                    color: step === stepItem.id ? '#19E180' : Colors.sub,
                    fontSize: 10,
                    fontWeight: step === stepItem.id ? '700' : '500',
                  }}>
                    {stepItem.id}
                  </Text>
                </TouchableOpacity>
              ))}
            </ScrollView>
          </View>
        </View>
      </View>
      
      {/* Step Content */}
      {renderStepContent()}
      
      {/* Modals */}
      <LineItemModal
        visible={materialModal.visible}
        onClose={() => setMaterialModal({ visible: false, item: null })}
        item={materialModal.item}
        onSave={(item) => {
          if (materialModal.item) {
            setMaterialsCart(prev => prev.map(m => m.id === materialModal.item.id ? item : m));
          } else {
            setMaterialsCart(prev => [...prev, { ...item, id: Date.now().toString() }]);
          }
          setMaterialModal({ visible: false, item: null });
        }}
        title="Material"
        laborMode={false}
      />
      
      <LineItemModal
        visible={laborModal.visible}
        onClose={() => setLaborModal({ visible: false, item: null })}
        item={laborModal.item}
        onSave={(item) => {
          const updated = bid.laborLineItems || [];
          if (laborModal.item) {
            const index = updated.findIndex(l => l.id === laborModal.item.id);
            if (index >= 0) {
              updated[index] = item;
            }
          } else {
            updated.push({ ...item, id: Date.now().toString() });
          }
          updateBid('laborLineItems', updated);
          setLaborModal({ visible: false, item: null });
        }}
        title="Labor"
        laborMode={true}
      />
      
      {/* Payment Milestone Modal */}
      <PaymentMilestoneModal
        visible={milestoneModal.visible}
        onClose={() => setMilestoneModal({ visible: false, item: null })}
        item={milestoneModal.item}
        onSave={handleSaveMilestone}
        grandTotal={calc?.grandTotal || calc?.total || 0}
      />
      
      {/* Weekly Payment Modal */}
      <WeeklyPaymentModal
        visible={weeklyPaymentModal.visible}
        onClose={() => setWeeklyPaymentModal({ visible: false, item: null })}
        item={weeklyPaymentModal.item}
        onSave={handleSaveWeeklyPayment}
        grandTotal={calc?.grandTotal || calc?.total || 0}
      />
      
      {/* Recovery Modal - Full Page */}
      <Modal
        visible={showRecoveryModal}
        transparent={false}
        animationType="slide"
        onRequestClose={() => setShowRecoveryModal(false)}
      >
        <SafeAreaView style={{ flex: 1, backgroundColor: '#000000' }}>
          <StatusBar barStyle="light-content" />
          <ScrollView
            contentContainerStyle={{ paddingTop: 20, paddingHorizontal: 20, paddingBottom: 120 }}
            showsVerticalScrollIndicator={false}
          >
            {/* Header */}
            <View style={{
              flexDirection: 'row',
              alignItems: 'flex-start',
              marginTop: 12,
              marginBottom: 18,
            }}>
              <View style={{ flex: 1 }}>
                <View style={{ flexDirection: 'row', alignItems: 'center', marginBottom: 4 }}>
                  <TouchableOpacity
                    onPress={() => {
                      Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Medium);
                      setShowRecoveryModal(false);
                    }}
                    style={{
                      width: 40,
                      height: 40,
                      borderRadius: 20,
                      borderWidth: 1,
                      backgroundColor: 'rgba(34, 197, 94, 0.2)',
                      borderColor: 'rgba(34, 197, 94, 0.3)',
                      justifyContent: 'center',
                      alignItems: 'center',
                      marginRight: 12,
                    }}
                  >
                    <MaterialIcons name="arrow-back" size={24} color="#22c55e" />
                  </TouchableOpacity>
                  <Text style={{ color: '#f9fafb', fontSize: 32, fontWeight: '800' }}>
                    Restore
                  </Text>
                </View>
                <Text style={{ color: '#9ca3af', fontSize: 14, marginTop: 4, marginLeft: 52 }}>
                  {savedEstimates.length} {savedEstimates.length === 1 ? 'saved bid' : 'saved bids'}
                </Text>
              </View>
            </View>
            
            {/* List of Saved Estimates */}
            {savedEstimates.length === 0 ? (
              <View style={{
                padding: 48,
                alignItems: 'center',
                justifyContent: 'center',
              }}>
                <View style={{
                  width: 80,
                  height: 80,
                  borderRadius: 40,
                  backgroundColor: 'rgba(34, 197, 94, 0.1)',
                  justifyContent: 'center',
                  alignItems: 'center',
                  marginBottom: 24,
                }}>
                  <Ionicons name="document-outline" size={40} color="#22c55e" />
                </View>
                <Text style={{ color: '#f9fafb', fontSize: 18, fontWeight: '700', marginBottom: 8, textAlign: 'center' }}>
                  No saved bids found
                </Text>
                <Text style={{ color: '#9ca3af', fontSize: 14, textAlign: 'center' }}>
                  Save a bid to restore it later
                </Text>
              </View>
            ) : (
              <View style={{
                borderRadius: 28,
                padding: 18,
                backgroundColor: '#020617',
                borderWidth: 1,
                borderColor: 'rgba(34, 197, 94, 0.25)',
                marginBottom: 16,
                shadowColor: '#000',
                shadowOpacity: 0.4,
                shadowRadius: 18,
                shadowOffset: { width: 0, height: 10 },
              }}>
                <View style={{ marginBottom: 16 }}>
                  <Text style={{ color: '#FFFFFF', fontSize: 20, fontWeight: '700', marginBottom: 2 }}>
                    Saved Bids
                  </Text>
                  <Text style={{ marginTop: 2, fontSize: 13, color: '#8DA0B8' }}>
                    Tap to restore a saved bid
                  </Text>
                </View>
                
                {savedEstimates.map((item, index) => (
                  <TouchableOpacity
                    key={item.id}
                    style={{
                      borderRadius: 24,
                      padding: 16,
                      marginBottom: index < savedEstimates.length - 1 ? 12 : 0,
                      backgroundColor: 'transparent',
                      borderWidth: 1,
                      borderColor: 'rgba(34, 197, 94, 0.3)',
                      shadowColor: '#000',
                      shadowOpacity: 0.4,
                      shadowRadius: 18,
                      shadowOffset: { width: 0, height: 10 },
                    }}
                    onPress={() => {
                      Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light);
                      loadEstimate(item);
                      setShowRecoveryModal(false);
                    }}
                    activeOpacity={0.7}
                  >
                    <LinearGradient
                      colors={["rgba(34, 197, 94, 0.15)", "rgba(34, 211, 238, 0.08)", "rgba(34, 197, 94, 0)"]}
                      start={{ x: 0, y: 0 }}
                      end={{ x: 1, y: 1 }}
                      style={{
                        borderRadius: 24,
                        padding: 16,
                        position: 'absolute',
                        top: 0,
                        left: 0,
                        right: 0,
                        bottom: 0,
                      }}
                    />
                    <View style={{ flexDirection: 'row', justifyContent: 'space-between', alignItems: 'flex-start', position: 'relative', zIndex: 1 }}>
                      <View style={{ flex: 1, marginRight: 12 }}>
                        <Text style={{ color: '#FFFFFF', fontSize: 18, fontWeight: '700', marginBottom: 8 }}>
                          {item.title || 'Untitled Bid'}
                        </Text>
                        <View style={{ flexDirection: 'row', alignItems: 'center', marginBottom: 6 }}>
                          <Ionicons name="person-outline" size={14} color="#8DA0B8" style={{ marginRight: 6 }} />
                          <Text style={{ color: '#7C8BA0', fontSize: 13 }}>
                            {item.customer || 'Unknown Customer'}
                          </Text>
                        </View>
                        <View style={{ flexDirection: 'row', alignItems: 'center' }}>
                          <Ionicons name="calendar-outline" size={14} color="#8DA0B8" style={{ marginRight: 6 }} />
                          <Text style={{ color: '#7C8BA0', fontSize: 13 }}>
                            {item.timestamp ? new Date(item.timestamp).toLocaleDateString() : 'Unknown date'}
                          </Text>
                        </View>
                      </View>
                      <View style={{ alignItems: 'flex-end' }}>
                        <Text style={{ color: '#22c55e', fontSize: 22, fontWeight: '700', marginBottom: 12 }}>
                          ${(item.total || 0).toLocaleString()}
                        </Text>
                        <TouchableOpacity
                          style={{
                            paddingHorizontal: 12,
                            paddingVertical: 8,
                            borderRadius: 12,
                            backgroundColor: 'rgba(239, 68, 68, 0.1)',
                            borderWidth: 1,
                            borderColor: 'rgba(239, 68, 68, 0.2)',
                          }}
                          onPress={(e) => {
                            e.stopPropagation();
                            Alert.alert(
                              'Delete Bid?',
                              `Are you sure you want to delete "${item.title || 'Untitled Bid'}"?`,
                              [
                                { text: 'Cancel', style: 'cancel' },
                                {
                                  text: 'Delete',
                                  style: 'destructive',
                                  onPress: () => deleteEstimate(item.id),
                                },
                              ]
                            );
                          }}
                        >
                          <Ionicons name="trash-outline" size={18} color="#ef4444" />
                        </TouchableOpacity>
                      </View>
                    </View>
                  </TouchableOpacity>
                ))}
              </View>
            )}
          </ScrollView>
        </SafeAreaView>
      </Modal>
    </SafeAreaView>
  );
}
